import { Component, signal,  } from '@angular/core';

@Component({ 
  standalone: true, 
  selector: app-TickerComponent,
  template: `<p>{{ price }}</p>`
})
export class TickerComponent implements OnInit {
price = signal(0);
constructor(private api: PriceService) {}
ngOnInit() {
this.api.stream$.subscribe(p => this.price = p); // live price feed
}
}
